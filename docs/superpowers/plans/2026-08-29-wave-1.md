# Wave 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix subagent attribution, recover lost task attribution via correlation backfill, make attribution rate visible in doctor, and persist overlap findings as real events so the feedback loop has input.

**Architecture:** Three independent slices: (1) nest subagent spans in `soleCallCovering` by detecting descendant agent IDs, (2) backfill null-attribution events by propagating single-task correlations and add an attribution metric to doctor, (3) persist contention findings from overlap computation into the ledger as `finding.created` events.

**Tech Stack:** TypeScript, SQLite (node:sqlite), existing `patchmesh-protocol` and `patchmesh-analyzers` packages.

## Global Constraints

- Node.js 24 or newer (required for `node:sqlite`)
- Hooks always exit 0 — a broken PatchMesh never breaks a session
- Warn, never block — PatchMesh may inject text, never deny/pause/redirect tool calls
- Events are append-only — never mutate stored events
- Do not store secrets, tokens, or credentials
- Every change ships green via `pnpm check`

```typescript
function isDescendant(childId: string, parentId: string): boolean {
  return childId.startsWith(`${parentId}.`);
}

function nestedCallCovering(
  worktreeRoot: string,
  change: ObservedFileChange,
  calls: readonly EffectAttributionCall[],
): EffectAttributionCall | null {
  const at = changeTimeMs(worktreeRoot, change);
  if (at === null) return null;
  const covering = calls.filter(
    (call) => at >= call.startedAtMs && at <= call.completedAtMs,
  );
  if (covering.length === 0) return null;
  if (covering.length === 1) return covering[0]!;
  // Multiple calls cover this mtime. Prefer a child over its parent.
  for (const candidate of covering) {
    if (candidate.agentId === null) continue;
    const isChild = covering.some(
      (other) =>
        other !== candidate &&
        other.agentId !== null &&
        isDescendant(candidate.agentId, other.agentId),
    );
    if (isChild) return candidate;
  }
  return null;
}
```

```typescript
/**
 * Propagate task attribution within correlation groups that have exactly one task.
 *
 * Where a correlation has exactly one non-null taskId among its events, that task is
 * assigned to every null-attribution event in the group. Groups with zero tasks or
 * multiple tasks stay unchanged — the former is honest absence, the latter is
 * ambiguous.
 */
export function backfillAttribution(ledgerPath: string): void {
  const store = SqliteEventStore.open(ledgerPath);
  try {
    // Find correlation groups with exactly one distinct non-null task
    const groups = store.db.prepare(`
      SELECT correlationId, GROUP_CONCAT(DISTINCT taskId) as tasks
      FROM events
      WHERE correlationId IS NOT NULL
      GROUP BY correlationId
      HAVING COUNT(DISTINCT taskId) = 1
         AND MAX(taskId) IS NOT NULL
    `).all() as Array<{ correlationId: string; tasks: string }>;

    for (const group of groups) {
      const task = group.tasks;
      store.db.prepare(`
        UPDATE events SET taskId = ?
        WHERE correlationId = ? AND taskId IS NULL
      `).run(task, group.correlationId);
    }
  } finally {
    store.close();
  }
}
```

```typescript
function checkAttribution(worktreeRoot: string): DoctorCheck {
  const ledgerPath = join(worktreeRoot, ".patchmesh", "ledger.db");
  if (!existsSync(ledgerPath)) {
    return { name: "attribution", status: "warn", detail: "no ledger — attribution unknown" };
  }
  try {
    const store = SqliteEventStore.open(ledgerPath);
    try {
      const total = store.count();
      if (total === 0) {
        return { name: "attribution", status: "ok", detail: "0 events — nothing to attribute" };
      }
      const attributed = store.db.prepare(
        "SELECT COUNT(*) as n FROM events WHERE taskId IS NOT NULL"
      ).get() as { n: number };
      const rate = Math.round((attributed.n / total) * 100);
      return {
        name: "attribution",
        status: rate >= 80 ? "ok" : "warn",
        detail: `${rate}% of events carry a task (${attributed.n}/${total})`,
      };
    } finally {
      store.close();
    }
  } catch {
    return { name: "attribution", status: "warn", detail: "could not read ledger" };
  }
}
```

```typescript
import { createHash } from "node:crypto";
import type { EventId, FindingCreatedEvent } from "patchmesh-protocol";

function findingId(overlap: ResourceOverlap): string {
  const key = JSON.stringify({
    path: overlap.logicalPath,
    agents: overlap.tasks.map((t) => t.agentId).sort(),
    earlier: overlap.contention.earlierWriteAt,
    later: overlap.contention.laterWriteAt,
  });
  return `finding_${createHash("sha256").update(key).digest("hex").slice(0, 32)}`;
}

export function persistFindings(
  ledgerPath: string,
  overlaps: readonly ResourceOverlap[],
): void {
  if (overlaps.length === 0) return;
  const store = SqliteEventStore.open(ledgerPath);
  try {
    // Read existing finding IDs for dedup
    const existing = new Set(
      store.read({ eventTypes: ["finding.created"] })
        .map((e: any) => e.payload.findingId)
    );

    const now = new Date().toISOString();
    for (const overlap of overlaps) {
      const id = findingId(overlap);
      if (existing.has(id)) continue;

      const event: FindingCreatedEvent = {
        schemaVersion: 1,
        eventId: `evt_${createHash("sha256").update(`${id}${now}`).digest("hex").slice(0, 32)}` as EventId,
        eventType: "finding.created",
        source: { kind: "watcher", sourceId: "source_patchmesh_overlap" },
        timestamp: now,
        repositoryId: "", // filled by store
        workspaceId: "" as any,
        worktreeId: "" as any,
        agentId: null,
        taskId: null,
        correlationId: "" as any,
        causationId: null,
        sourceSequence: null,
        payload: {
          findingId: id,
          type: "contention",
          filePath: overlap.logicalPath,
          agentIds: overlap.tasks.map((t) => t.agentId),
          evidenceEventIds: [],
          createdAt: now,
        },
      };
      store.appendAtomic([event]);
    }
  } finally {
    store.close();
  }
}
```
